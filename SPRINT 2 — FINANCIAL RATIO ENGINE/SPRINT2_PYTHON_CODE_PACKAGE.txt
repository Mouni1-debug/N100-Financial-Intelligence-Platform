"""Input validation for ratio computation."""

from typing import Optional, Tuple
import logging

logger = logging.getLogger(__name__)


class RatioInputValidator:
    """Validate inputs before ratio computation."""
    
    @staticmethod
    def validate_profitability_inputs(
        net_profit: Optional[float],
        sales: Optional[float],
        operating_profit: Optional[float],
        equity: Optional[float],
        assets: Optional[float]
    ) -> Tuple[bool, str]:
        """
        Validate profitability ratio inputs.
        
        Returns:
            (is_valid, error_message)
        """
        if net_profit is not None and not isinstance(net_profit, (int, float)):
            return False, "net_profit must be numeric"
        
        if sales is not None and sales < 0:
            return False, "sales cannot be negative"
        
        return True, ""
    
    @staticmethod
    def validate_leverage_inputs(
        borrowings: Optional[float],
        equity: Optional[float],
        interest_expense: Optional[float]
    ) -> Tuple[bool, str]:
        """Validate leverage ratio inputs."""
        
        if borrowings is not None and borrowings < 0:
            return False, "borrowings cannot be negative"
        
        if interest_expense is not None and interest_expense < 0:
            return False, "interest_expense cannot be negative"
        
        return True, ""

"""Handle edge cases for financial companies."""

import logging
from typing import Optional, Dict

logger = logging.getLogger(__name__)

# Financial sector company identifiers
FINANCIAL_SECTOR_COMPANIES = {
    'HDFCBANK', 'ICICIBANK', 'AXISBANK', 'KOTAKBANK',  # Banks
    'SBIN', 'INDUSINDBK', 'FEDERALBNK',  # Public Banks
    'HDFC', 'INDIABULL', 'EQUITAS',  # Finance companies
    'RELIANCE',  # Has finance subsidiary
    'BAJAJFINSV', 'BAJAJHLDNG',  # Finance/Insurance
}


class EdgeCaseHandler:
    """Handle special cases for financial companies and anomalies."""
    
    @staticmethod
    def suppress_leverage_flag_for_financials(
        company_id: str,
        high_leverage_flag: Optional[bool],
        debt_to_equity: Optional[float]
    ) -> Optional[bool]:
        """
        For financial companies, suppress high_leverage_flag.
        
        Banks operate with 5-10x leverage by design, which is normal.
        Non-financial companies with high leverage are concerning.
        
        Args:
            company_id: Company identifier
            high_leverage_flag: Original high leverage flag
            debt_to_equity: D/E ratio
        
        Returns:
            None for financial companies, original flag otherwise
        """
        if company_id in FINANCIAL_SECTOR_COMPANIES:
            # Don't flag financial companies for high leverage
            return None
        
        return high_leverage_flag
    
    @staticmethod
    def cross_check_roce(
        computed_roce: Optional[float],
        source_roce: Optional[float],
        company_id: str,
        fiscal_year: int
    ) -> Dict:
        """
        Cross-check computed ROCE against source data.
        
        Returns dict with:
        - is_anomaly: bool
        - difference: float
        - status: "ACCEPTABLE" or "ANOMALY"
        - comment: str
        """
        if computed_roce is None or source_roce is None:
            return {
                'is_anomaly': False,
                'difference': None,
                'status': 'NO_DATA',
                'comment': 'Missing computed or source ROCE'
            }
        
        difference = abs(computed_roce - source_roce)
        
        is_anomaly = difference > 5.0  # Threshold: 5% difference
        
        return {
            'is_anomaly': is_anomaly,
            'difference': round(difference, 2),
            'status': 'ANOMALY' if is_anomaly else 'ACCEPTABLE',
            'comment': f"Difference of {round(difference, 2)}%"
        }
    
    @staticmethod
    def cross_check_roe(
        computed_roe: Optional[float],
        source_roe: Optional[float],
        company_id: str,
        fiscal_year: int
    ) -> Dict:
        """
        Cross-check computed ROE against source data.
        
        Note: Some source ROE values appear incorrect (e.g., TCS shows 0.52, likely should be 52).
        Use computed ROE for analytics, flag source anomalies.
        """
        if computed_roe is None or source_roe is None:
            return {
                'is_anomaly': False,
                'difference': None,
                'status': 'NO_DATA',
                'comment': 'Missing computed or source ROE'
            }
        
        difference = abs(computed_roe - source_roe)
        is_anomaly = difference > 5.0
        
        return {
            'is_anomaly': is_anomaly,
            'difference': round(difference, 2),
            'status': 'ANOMALY' if is_anomaly else 'ACCEPTABLE',
            'comment': f"Source ROE: {source_roe}, Computed: {computed_roe}",
            'recommendation': 'Use computed ROE for analytics'
        }
