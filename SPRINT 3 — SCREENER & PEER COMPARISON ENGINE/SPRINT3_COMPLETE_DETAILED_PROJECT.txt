"""Filter Engine - Core screening logic."""

import pandas as pd
import numpy as np
import yaml
from pathlib import Path
from typing import Optional, Dict, List, Tuple
import logging

logger = logging.getLogger(__name__)


class ScreenerEngine:
    """
    Core screener engine for filtering companies based on 15 metrics.
    
    Day 15 Implementation:
    - Loads screener_config.yaml
    - Applies threshold filters
    - Handles special cases (D/E for Financials, ICR for Debt Free)
    - Returns sorted DataFrame with composite_quality_score
    """
    
    def __init__(self, config_path: str = "config/screener_config.yaml"):
        """Initialize screener engine with config."""
        self.config_path = Path(config_path)
        self.config = self._load_config()
        self.filter_config = self.config.get('screener_config', {}).get('filters', {})
        self.exclusions = self.config.get('screener_config', {}).get('exclusions', {})
        logger.info(f"ScreenerEngine initialized with config: {config_path}")
    
    def _load_config(self) -> Dict:
        """Load YAML configuration file."""
        try:
            with open(self.config_path, 'r') as f:
                return yaml.safe_load(f)
        except FileNotFoundError:
            logger.error(f"Config file not found: {self.config_path}")
            raise
    
    def apply_filter(self, df: pd.DataFrame, metric: str, operator: str, threshold: float) -> pd.DataFrame:
        """
        Apply single filter to DataFrame.
        
        Args:
            df: Input DataFrame with financial data
            metric: Column name to filter on
            operator: 'min' or 'max'
            threshold: Threshold value
            
        Returns:
            Filtered DataFrame
            
        Examples:
            >>> screener.apply_filter(df, 'return_on_equity_pct', 'min', 15)
            >>> screener.apply_filter(df, 'debt_to_equity', 'max', 2.0)
        """
        if metric not in df.columns:
            logger.warning(f"Metric {metric} not found in DataFrame columns")
            return df
        
        # Handle null values
        df = df.copy()
        
        if operator == 'min':
            filtered = df[df[metric] >= threshold]
        elif operator == 'max':
            filtered = df[df[metric] <= threshold]
        else:
            logger.error(f"Unknown operator: {operator}")
            return df
        
        logger.info(f"Filter {metric} {operator} {threshold}: {len(filtered)} companies pass")
        return filtered
    
    def apply_roe_filter(self, df: pd.DataFrame, min_roe: float) -> pd.DataFrame:
        """Apply ROE minimum filter."""
        filtered = df[df['return_on_equity_pct'].fillna(0) >= min_roe].copy()
        logger.info(f"ROE >= {min_roe}%: {len(filtered)} companies pass")
        return filtered
    
    def apply_de_filter(self, df: pd.DataFrame, max_de: float) -> pd.DataFrame:
        """
        Apply D/E maximum filter with sector exclusion.
        
        Special handling: Skip Financials sector (structural high leverage is normal)
        """
        df = df.copy()
        sectors_to_skip = self.exclusions.get('sectors_for_de_filter', ['Financials'])
        
        # Separate into Financials and non-Financials
        is_financial = df['broad_sector'].isin(sectors_to_skip)
        non_financial = df[~is_financial]
        financial = df[is_financial]
        
        # Apply D/E filter only to non-Financials
        non_financial_filtered = non_financial[non_financial['debt_to_equity'].fillna(0) <= max_de]
        
        # Combine results
        filtered = pd.concat([non_financial_filtered, financial], ignore_index=True)
        logger.info(f"D/E <= {max_de}: {len(filtered)} companies pass (Financials excluded from filter)")
        return filtered
    
    def apply_fcf_filter(self, df: pd.DataFrame, min_fcf: float) -> pd.DataFrame:
        """Apply Free Cash Flow minimum filter."""
        filtered = df[df['free_cash_flow_cr'].fillna(0) >= min_fcf].copy()
        logger.info(f"FCF >= {min_fcf}Cr: {len(filtered)} companies pass")
        return filtered
    
    def apply_revenue_cagr_5yr_filter(self, df: pd.DataFrame, min_cagr: float) -> pd.DataFrame:
        """Apply 5-year Revenue CAGR minimum filter."""
        filtered = df[df['revenue_cagr_5yr_pct'].fillna(0) >= min_cagr].copy()
        logger.info(f"Revenue CAGR 5yr >= {min_cagr}%: {len(filtered)} companies pass")
        return filtered
    
    def apply_pat_cagr_5yr_filter(self, df: pd.DataFrame, min_cagr: float) -> pd.DataFrame:
        """Apply 5-year PAT CAGR minimum filter."""
        filtered = df[df['pat_cagr_5yr_pct'].fillna(0) >= min_cagr].copy()
        logger.info(f"PAT CAGR 5yr >= {min_cagr}%: {len(filtered)} companies pass")
        return filtered
    
    def apply_opm_filter(self, df: pd.DataFrame, min_opm: float) -> pd.DataFrame:
        """Apply OPM minimum filter."""
        filtered = df[df['operating_profit_margin_pct'].fillna(0) >= min_opm].copy()
        logger.info(f"OPM >= {min_opm}%: {len(filtered)} companies pass")
        return filtered
    
    def apply_pe_filter(self, df: pd.DataFrame, max_pe: float) -> pd.DataFrame:
        """Apply P/E maximum filter."""
        filtered = df[df['pe_ratio'].fillna(999) <= max_pe].copy()
        logger.info(f"P/E <= {max_pe}: {len(filtered)} companies pass")
        return filtered
    
    def apply_pb_filter(self, df: pd.DataFrame, max_pb: float) -> pd.DataFrame:
        """Apply P/B maximum filter."""
        filtered = df[df['pb_ratio'].fillna(999) <= max_pb].copy()
        logger.info(f"P/B <= {max_pb}: {len(filtered)} companies pass")
        return filtered
    
    def apply_dividend_yield_filter(self, df: pd.DataFrame, min_yield: float) -> pd.DataFrame:
        """Apply Dividend Yield minimum filter."""
        filtered = df[df['dividend_yield_pct'].fillna(0) >= min_yield].copy()
        logger.info(f"Dividend Yield >= {min_yield}%: {len(filtered)} companies pass")
        return filtered
    
    def apply_icr_filter(self, df: pd.DataFrame, min_icr: float) -> pd.DataFrame:
        """
        Apply ICR minimum filter with special handling for Debt Free companies.
        
        Special handling: Companies with ICR label "Debt Free" are treated as 
        ICR = infinity (always pass any minimum threshold)
        """
        df = df.copy()
        
        # Mark Debt Free companies as passing
        is_debt_free = df['icr_label'] == 'Debt Free'
        
        # Apply ICR filter to non-debt-free companies
        passes_icr = df['interest_coverage_ratio'].fillna(0) >= min_icr
        
        # Combine: Debt Free OR passes ICR
        filtered = df[is_debt_free | passes_icr].copy()
        logger.info(f"ICR >= {min_icr}: {len(filtered)} companies pass (Debt Free always pass)")
        return filtered
    
    def apply_market_cap_filter(self, df: pd.DataFrame, min_cap: float) -> pd.DataFrame:
        """Apply Market Cap minimum filter."""
        filtered = df[df['market_cap_cr'].fillna(0) >= min_cap].copy()
        logger.info(f"Market Cap >= {min_cap}Cr: {len(filtered)} companies pass")
        return filtered
    
    def apply_net_profit_filter(self, df: pd.DataFrame, min_profit: float) -> pd.DataFrame:
        """Apply Net Profit minimum filter."""
        filtered = df[df['net_profit_cr'].fillna(0) >= min_profit].copy()
        logger.info(f"Net Profit >= {min_profit}Cr: {len(filtered)} companies pass")
        return filtered
    
    def apply_eps_cagr_filter(self, df: pd.DataFrame, min_cagr: float) -> pd.DataFrame:
        """Apply EPS CAGR minimum filter."""
        filtered = df[df['eps_cagr_5yr_pct'].fillna(0) >= min_cagr].copy()
        logger.info(f"EPS CAGR >= {min_cagr}%: {len(filtered)} companies pass")
        return filtered
    
    def apply_asset_turnover_filter(self, df: pd.DataFrame, min_turnover: float) -> pd.DataFrame:
        """Apply Asset Turnover minimum filter."""
        filtered = df[df['asset_turnover'].fillna(0) >= min_turnover].copy()
        logger.info(f"Asset Turnover >= {min_turnover}: {len(filtered)} companies pass")
        return filtered
    
    def apply_sales_filter(self, df: pd.DataFrame, min_sales: float) -> pd.DataFrame:
        """Apply Sales (Revenue) minimum filter."""
        filtered = df[df['sales_cr'].fillna(0) >= min_sales].copy()
        logger.info(f"Sales >= {min_sales}Cr: {len(filtered)} companies pass")
        return filtered
    
    def screen(self, df: pd.DataFrame, filters: Optional[Dict] = None) -> pd.DataFrame:
        """
        Apply all filters and return screened results.
        
        Args:
            df: Input DataFrame with financial data
            filters: Optional dict of filters to apply
                     If None, uses all filters from config
                     
        Returns:
            Filtered and sorted DataFrame
        """
        if filters is None:
            filters = self.filter_config
        
        logger.info(f"Starting screening with {len(df)} companies")
        result = df.copy()
        
        # Apply each filter
        if 'roe_min' in filters and filters['roe_min'] > 0:
            result = self.apply_roe_filter(result, filters['roe_min'])
        
        if 'de_max' in filters:
            result = self.apply_de_filter(result, filters['de_max'])
        
        if 'fcf_min' in filters and filters['fcf_min'] > 0:
            result = self.apply_fcf_filter(result, filters['fcf_min'])
        
        if 'revenue_cagr_5yr_min' in filters and filters['revenue_cagr_5yr_min'] > 0:
            result = self.apply_revenue_cagr_5yr_filter(result, filters['revenue_cagr_5yr_min'])
        
        if 'pat_cagr_5yr_min' in filters and filters['pat_cagr_5yr_min'] > 0:
            result = self.apply_pat_cagr_5yr_filter(result, filters['pat_cagr_5yr_min'])
        
        if 'opm_min' in filters and filters['opm_min'] > 0:
            result = self.apply_opm_filter(result, filters['opm_min'])
        
        if 'pe_max' in filters:
            result = self.apply_pe_filter(result, filters['pe_max'])
        
        if 'pb_max' in filters:
            result = self.apply_pb_filter(result, filters['pb_max'])
        
        if 'dividend_yield_min' in filters and filters['dividend_yield_min'] >= 0:
            result = self.apply_dividend_yield_filter(result, filters['dividend_yield_min'])
        
        if 'icr_min' in filters and filters['icr_min'] > 0:
            result = self.apply_icr_filter(result, filters['icr_min'])
        
        if 'market_cap_min' in filters and filters['market_cap_min'] > 0:
            result = self.apply_market_cap_filter(result, filters['market_cap_min'])
        
        if 'net_profit_min' in filters and filters['net_profit_min'] > 0:
            result = self.apply_net_profit_filter(result, filters['net_profit_min'])
        
        if 'eps_cagr_min' in filters and filters['eps_cagr_min'] > 0:
            result = self.apply_eps_cagr_filter(result, filters['eps_cagr_min'])
        
        if 'asset_turnover_min' in filters and filters['asset_turnover_min'] > 0:
            result = self.apply_asset_turnover_filter(result, filters['asset_turnover_min'])
        
        if 'sales_min' in filters and filters['sales_min'] > 0:
            result = self.apply_sales_filter(result, filters['sales_min'])
        
        # Sort by composite_quality_score descending
        if 'composite_quality_score' in result.columns:
            result = result.sort_values('composite_quality_score', ascending=False)
        
        logger.info(f"Screening complete: {len(result)} companies match criteria")
        return result

"""Preset screeners - 6 predefined investment strategies."""

import pandas as pd
from typing import Dict, Tuple
import logging

logger = logging.getLogger(__name__)


class ScreenerPresets:
    """
    Six preset screeners for different investment strategies.
    
    Day 16 Implementation:
    1. Quality Compounder: ROE>15%, D/E<1, FCF>0, Rev CAGR>10%
    2. Value Pick: P/E<20, P/B<3, D/E<2, Div Yield>1%
    3. Growth Accelerator: PAT CAGR>20%, Rev CAGR>15%, D/E<2
    4. Dividend Champion: Div Yield>2%, Payout<80%, FCF>0
    5. Debt-Free Blue Chip: D/E=0, ROE>12%, Rev>5000Cr
    6. Turnaround Watch: Rev CAGR 3yr>10%, FCF>0, D/E declining
    """
    
    def __init__(self, engine):
        """Initialize presets with screener engine reference."""
        self.engine = engine
    
    def quality_compounder(self, df: pd.DataFrame) -> Tuple[pd.DataFrame, Dict]:
        """
        Quality Compounder: High-quality compound growth.
        
        Criteria:
        - ROE > 15%
        - D/E < 1.0
        - FCF > 0
        - Revenue CAGR 5yr > 10%
        
        Expected: 10-30 companies
        """
        filters = {
            'roe_min': 15,
            'de_max': 1.0,
            'fcf_min': 0,
            'revenue_cagr_5yr_min': 10
        }
        
        result = self.engine.screen(df, filters)
        
        metadata = {
            'name': 'Quality Compounder',
            'description': 'High-quality companies with strong returns and controlled leverage',
            'criteria': filters,
            'expected_count': '10-30',
            'companies_found': len(result)
        }
        
        logger.info(f"Quality Compounder: {len(result)} companies found (expected 10-30)")
        return result, metadata
    
    def value_pick(self, df: pd.DataFrame) -> Tuple[pd.DataFrame, Dict]:
        """
        Value Pick: Undervalued with dividend support.
        
        Criteria:
        - P/E < 20
        - P/B < 3.0
        - D/E < 2.0
        - Dividend Yield > 1%
        
        Expected: 15-40 companies
        """
        filters = {
            'pe_max': 20,
            'pb_max': 3.0,
            'de_max': 2.0,
            'dividend_yield_min': 1
        }
        
        result = self.engine.screen(df, filters)
        
        metadata = {
            'name': 'Value Pick',
            'description': 'Undervalued companies with reasonable dividend yields',
            'criteria': filters,
            'expected_count': '15-40',
            'companies_found': len(result)
        }
        
        logger.info(f"Value Pick: {len(result)} companies found (expected 15-40)")
        return result, metadata
    
    def growth_accelerator(self, df: pd.DataFrame) -> Tuple[pd.DataFrame, Dict]:
        """
        Growth Accelerator: Fast-growing with controlled leverage.
        
        Criteria:
        - PAT CAGR 5yr > 20%
        - Revenue CAGR 5yr > 15%
        - D/E < 2.0
        
        Expected: 5-20 companies

            'pat_cagr_5yr_min': 20,
            'revenue_cagr_5yr_min': 15,
            'de_max': 2.0

        Dividend Champion: Consistent high dividend payers.
        
        Criteria:

"""Composite Quality Score Calculator - 0-100 scale."""

import pandas as pd
import numpy as np
from typing import Dict, Tuple
import logging

logger = logging.getLogger(__name__)


class CompositeScoreCalculator:
    """
    Calculate composite quality score (0-100 scale).
    
    Day 17 Implementation:
    - 35% Profitability: ROE(15%) + ROCE(10%) + NPM(10%)
    - 30% Cash Quality: FCF CAGR(15%) + CFO/PAT(10%) + FCF flag(5%)
    - 20% Growth: Revenue CAGR 5yr(10%) + PAT CAGR 5yr(10%)
    - 15% Leverage: D/E score(10%) + ICR score(5%)
    
    Uses P10/P90 winsorisation for normalization.
    """
    
    def __init__(self, config: Dict):
        """Initialize with weight configuration."""
        self.config = config.get('screener_config', {}).get('composite_score', {})
        self.weights = {
            'profitability': self.config.get('profitability_weight', 0.35),
            'cash_quality': self.config.get('cash_quality_weight', 0.30),
            'growth': self.config.get('growth_weight', 0.20),
            'leverage': self.config.get('leverage_weight', 0.15)
        }
    
    def _winsorize_p10_p90(self, series: pd.Series) -> pd.Series:
        """Apply P10/P90 winsorisation to cap extreme values."""
        p10 = series.quantile(0.10)
        p90 = series.quantile(0.90)
        return series.clip(lower=p10, upper=p90)
    
    def _normalize_to_0_100(self, series: pd.Series) -> pd.Series:
        """Normalize series to 0-100 scale."""
        s_min = series.min()
        s_max = series.max()
        
        if s_min == s_max:
            return pd.Series([50] * len(series), index=series.index)
        
        return ((series - s_min) / (s_max - s_min)) * 100
    
    def calculate_profitability_score(self, df: pd.DataFrame) -> pd.Series:
        """
        Calculate profitability component score.
        
        Components:
        - ROE (15%): Return on Equity
        - ROCE (10%): Return on Capital Employed
        - NPM (10%): Net Profit Margin
        """
        roe = df['return_on_equity_pct'].fillna(0)
        roce = df['return_on_capital_employed_pct'].fillna(0)
        npm = df['net_profit_margin_pct'].fillna(0)
        
        # Winsorize and normalize each
        roe_norm = self._normalize_to_0_100(self._winsorize_p10_p90(roe))
        roce_norm = self._normalize_to_0_100(self._winsorize_p10_p90(roce))
        npm_norm = self._normalize_to_0_100(self._winsorize_p10_p90(npm))
        
        # Weighted average
        score = (roe_norm * 0.15 + roce_norm * 0.10 + npm_norm * 0.10) / 0.35
        
        logger.info("Profitability score calculated")
        return score
    
    def calculate_cash_quality_score(self, df: pd.DataFrame) -> pd.Series:
        """
        Calculate cash quality component score.
        
        Components:
        - FCF CAGR (15%): Free Cash Flow CAGR
        - CFO/PAT ratio (10%): Operating Cash Flow to Net Profit ratio
        - FCF positive flag (5%): Whether FCF is positive (binary)
        """
        fcf_cagr = df['fcf_cagr_5yr_pct'].fillna(0)
        cfo_pat = df['cfo_quality_score'].fillna(0.5)
        fcf_positive = (df['free_cash_flow_cr'].fillna(0) > 0).astype(int) * 100
        
        # Winsorize and normalize
        fcf_cagr_norm = self._normalize_to_0_100(self._winsorize_p10_p90(fcf_cagr))
        cfo_pat_norm = self._normalize_to_0_100(self._winsorize_p10_p90(cfo_pat * 100))
        
        # Weighted average
        score = (fcf_cagr_norm * 0.15 + cfo_pat_norm * 0.10 + fcf_positive * 0.05) / 0.30
        
        logger.info("Cash quality score calculated")
        return score
    
    def calculate_growth_score(self, df: pd.DataFrame) -> pd.Series:
        """
        Calculate growth component score.
        
        Components:
        - Revenue CAGR 5yr (10%): Revenue growth rate
        - PAT CAGR 5yr (10%): Profit growth rate
        """
        rev_cagr = df['revenue_cagr_5yr_pct'].fillna(0)
        pat_cagr = df['pat_cagr_5yr_pct'].fillna(0)
        
        # Winsorize and normalize
        rev_cagr_norm = self._normalize_to_0_100(self._winsorize_p10_p90(rev_cagr))
        pat_cagr_norm = self._normalize_to_0_100(self._winsorize_p10_p90(pat_cagr))
        
        # Weighted average
        score = (rev_cagr_norm * 0.10 + pat_cagr_norm * 0.10) / 0.20
        
        logger.info("Growth score calculated")
        return score
    
    def calculate_leverage_score(self, df: pd.DataFrame) -> pd.Series:
        """
        Calculate leverage component score (inverted - lower is better).
        
        Components:
        - D/E score (10%): Debt to Equity (lower is better)
        - ICR score (5%): Interest Coverage Ratio (higher is better)
        """
        de = df['debt_to_equity'].fillna(0)
        icr = df['interest_coverage_ratio'].fillna(0)
        
        # For D/E: invert so lower D/E = higher score
        de_inverted = 1 / (1 + de)
        
        # Winsorize and normalize
        de_norm = self._normalize_to_0_100(self._winsorize_p10_p90(de_inverted))
        icr_norm = self._normalize_to_0_100(self._winsorize_p10_p90(icr))
        
        # Weighted average
        score = (de_norm * 0.10 + icr_norm * 0.05) / 0.15
        
        logger.info("Leverage score calculated")
        return score
    
    def calculate_composite_score(self, df: pd.DataFrame) -> pd.DataFrame:
        """
        Calculate final composite quality score (0-100).
        
        Combines all four component scores with their weights:
        - 35% Profitability
        - 30% Cash Quality
        - 20% Growth
        - 15% Leverage
        """
        df = df.copy()
        
        # Calculate component scores
        prof_score = self.calculate_profitability_score(df)
        cash_score = self.calculate_cash_quality_score(df)
        growth_score = self.calculate_growth_score(df)
        leverage_score = self.calculate_leverage_score(df)
        
        # Combine with weights
        df['composite_quality_score'] = (
            prof_score * self.weights['profitability'] +
            cash_score * self.weights['cash_quality'] +
            growth_score * self.weights['growth'] +
            leverage_score * self.weights['leverage']
        ).round(1)
        
        # Clamp to 0-100
        df['composite_quality_score'] = df['composite_quality_score'].clip(0, 100)
        
        logger.info(f"Composite scores calculated. Mean: {df['composite_quality_score'].mean():.1f}")
        return df
    
    def calculate_sector_relative_score(self, df: pd.DataFrame) -> pd.DataFrame:
        """
        Calculate sector-relative composite score.
        
        Normalizes composite score within each broad_sector so scores reflect
        performance vs sector peers.
        """
        df = df.copy()
        
        if 'broad_sector' not in df.columns:
            logger.warning("broad_sector not found, skipping sector-relative score")
            return df
        
        df['sector_relative_score'] = df.groupby('broad_sector')['composite_quality_score'].transform(
            lambda x: ((x - x.min()) / (x.max() - x.min()) * 100) if x.max() > x.min() else 50
        ).round(1)
        
        logger.info("Sector-relative scores calculated")
        return df

"""Data validation utilities for screener."""

import pandas as pd
import logging

logger = logging.getLogger(__name__)


class ScreenerValidator:
    """Validate screener inputs and outputs."""
    
    REQUIRED_COLUMNS = [
        'company_id', 'company_name', 'broad_sector',
        'return_on_equity_pct', 'debt_to_equity',
        'free_cash_flow_cr', 'revenue_cagr_5yr_pct',
        'pat_cagr_5yr_pct', 'operating_profit_margin_pct',
        'pe_ratio', 'pb_ratio', 'dividend_yield_pct',
        'interest_coverage_ratio', 'market_cap_cr',
        'net_profit_cr', 'eps_cagr_5yr_pct', 'asset_turnover',
        'sales_cr', 'icr_label'
    ]
    
    @classmethod
    def validate_input_dataframe(cls, df: pd.DataFrame) -> bool:
        """
        Validate that input DataFrame has all required columns.
        
        Returns:
            True if valid, False otherwise
        """
        missing_cols = set(cls.REQUIRED_COLUMNS) - set(df.columns)
        
        if missing_cols:
            logger.error(f"Missing required columns: {missing_cols}")
            return False
        
        if len(df) == 0:
            logger.error("Input DataFrame is empty")
            return False
        
        logger.info(f"Input validation passed: {len(df)} rows, {len(df.columns)} columns")
        return True
    
    @classmethod
    def validate_screener_output(cls, df: pd.DataFrame) -> bool:
        """
        Validate screener output.
        
        Returns:
            True if valid, False otherwise
        """
        required_for_output = cls.REQUIRED_COLUMNS + ['composite_quality_score']
        missing_cols = set(required_for_output) - set(df.columns)
        
        if missing_cols:
            logger.error(f"Missing output columns: {missing_cols}")
            return False
        
        # Check composite score is 0-100
        if df['composite_quality_score'].min() < 0 or df['composite_quality_score'].max() > 100:
            logger.warning("Composite scores outside 0-100 range")
        
        logger.info("Output validation passed")
        return True

class PeerRankingEngine:

    Day 18 Implementation:

    - Compute PERCENT_RANK for 10 metrics within each peer group
    - Populate peer_percentiles SQLite table

    1. ROE - Return on Equity (%)
    2. ROCE - Return on Capital Employed (%)
    3. NPM - Net Profit Margin (%)

    def __init__(self, db_path: str = "db/nifty100.db"):

    def calculate_percentile_rank(self, values: pd.Series, value: float) -> float:
        """
        Calculate percentile rank of a value within a series.
        
        Uses scipy.stats.percentileofscore with 'linear' interpolation.
        Returns 0-100 scale.
        
        Args:
            values: Series of values in peer group
            value: Value to rank
            
        Returns:
            Percentile rank 0-100
        """
        if pd.isna(value):
            return np.nan
        
        try:
            rank = percentileofscore(values.dropna(), value, kind='linear')
            return rank
        except Exception as e:
            logger.warning(f"Error calculating percentile: {e}")
            return np.nan
    
    def rank_metric_in_peer_group(self, 
                                   df: pd.DataFrame,
                                   peer_group: str,
                                   metric: str,
                                   peer_mapping: pd.DataFrame,
                                   invert: bool = False) -> pd.DataFrame:
        """
        Rank a single metric within a peer group.
        
        Args:
            df: Full DataFrame with financial data
            peer_group: Name of peer group (e.g., "IT Services")
            metric: Column name to rank (e.g., "return_on_equity_pct")
            peer_mapping: DataFrame with company_id and peer_group_name
            invert: If True, invert ranking (for metrics where lower is better)
            
        Returns:
            DataFrame with percentile_rank column
        """
        # Get companies in this peer group
        companies_in_group = peer_mapping[peer_mapping['peer_group_name'] == peer_group]['company_id'].tolist()
        
        if not companies_in_group:
            logger.warning(f"No companies found in peer group: {peer_group}")
            return pd.DataFrame()
        
        group_df = df[df['company_id'].isin(companies_in_group)].copy()
        
        if len(group_df) == 0:
            logger.warning(f"No data found for peer group: {peer_group}")
            return pd.DataFrame()
        
        # Get metric values for the group
        metric_values = group_df[metric].dropna()
        
        if len(metric_values) == 0:
            logger.warning(f"No values for metric {metric} in peer group {peer_group}")
            return group_df
        
        # Calculate percentile rank for each company
        def rank_func(val):
            if pd.isna(val):
                return np.nan
            
            # For D/E, invert ranking (lower D/E = higher percentile)
            if invert:
                # Invert the value: lower original = higher inverted = higher percentile
                inverted_val = 1 / (1 + val) if val >= 0 else 0
                inverted_values = 1 / (1 + metric_values.clip(lower=0))
                return percentileofscore(inverted_values, inverted_val, kind='linear')
            else:
                return percentileofscore(metric_values, val, kind='linear')
        
        group_df['percentile_rank'] = group_df[metric].apply(rank_func)
        
        return group_df[['company_id', 'company_name', metric, 'percentile_rank']]
    
    def compute_all_rankings(self, 
                            df: pd.DataFrame,
                            peer_mapping: pd.DataFrame) -> pd.DataFrame:
        """
        Compute percentile rankings for all metrics in all peer groups.
        
        Returns:
            DataFrame with columns: company_id, peer_group_name, metric, 
            value, percentile_rank, fiscal_year
        """
        results = []
        
        for peer_group in self.PEER_GROUPS:
            logger.info(f"Computing rankings for peer group: {peer_group}")
            
            for metric_col, metric_name in self.METRICS.items():
                # Check if metric column exists
                if metric_col not in df.columns:
                    logger.warning(f"Metric column not found: {metric_col}")
                    continue
                
                # Determine if this metric should be inverted
                invert = (metric_col == 'debt_to_equity')
                
                # Rank this metric in this peer group
                ranked = self.rank_metric_in_peer_group(df, peer_group, metric_col, peer_mapping, invert)
                
                if len(ranked) > 0:
                    ranked['peer_group_name'] = peer_group
                    ranked['metric'] = metric_name
                    ranked['fiscal_year'] = 2024  # Current year
                    results.append(ranked)
        
        if results:
            final_df = pd.concat(results, ignore_index=True)
            logger.info(f"Computed {len(final_df)} percentile rankings")
            return final_df
        else:
            return pd.DataFrame()
    
    def populate_peer_percentiles_table(self, 
                                       df: pd.DataFrame,
                                       peer_mapping: pd.DataFrame):
        """
        Populate peer_percentiles table in SQLite database.
        
        Expected schema:
        CREATE TABLE peer_percentiles (
            peer_id INTEGER PRIMARY KEY AUTOINCREMENT,
            company_id INTEGER NOT NULL,
            peer_group_name VARCHAR(100) NOT NULL,
            metric VARCHAR(100) NOT NULL,
            value DECIMAL(10,2),
            percentile_rank DECIMAL(5,2),
            fiscal_year INTEGER,
            created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
            UNIQUE(company_id, peer_group_name, metric, fiscal_year),
            FOREIGN KEY(company_id) REFERENCES companies(company_id)
        );
        """
        rankings = self.compute_all_rankings(df, peer_mapping)
        
        if len(rankings) == 0:
            logger.error("No rankings computed, aborting table population")
            return
        
        # Rename columns to match schema
        rankings_for_db = rankings.rename(columns={
            'return_on_equity_pct': 'value',
            'return_on_capital_employed_pct': 'value',
            'net_profit_margin_pct': 'value',
            'debt_to_equity': 'value',
            'free_cash_flow_cr': 'value',
            'pat_cagr_5yr_pct': 'value',
            'revenue_cagr_5yr_pct': 'value',
            'eps_cagr_5yr_pct': 'value',
            'interest_coverage_ratio': 'value',
            'asset_turnover': 'value'
        })
        
        # Write to SQLite
        try:
            rankings_for_db.to_sql('peer_percentiles', self.conn, if_exists='append', index=False)
            self.conn.commit()
            logger.info(f"Populated peer_percentiles table: {len(rankings_for_db)} rows")
        except Exception as e:
            logger.error(f"Error writing to database: {e}")
            self.conn.rollback()

"""Radar Chart Visualization Generator - Day 19 Implementation."""

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import matplotlib.patches as mpatches
from pathlib import Path
from typing import Optional, List
import logging

logger = logging.getLogger(__name__)


class RadarChartGenerator:
    """
    Generate radar/polar charts for company peer group comparison.
    
    Day 19 Implementation:
    - 8-axis radar chart: ROE, ROCE, NPM, D/E, FCF score, PAT CAGR 5yr, 
                          Revenue CAGR 5yr, Composite Score
    - Company values as filled polygon
    - Peer group average as dashed overlay
    - PNG export to reports/radar_charts/{company_id}_radar.png
    - For companies with no peer group, show vs Nifty 100 average
    
    Visualization settings:
    - 150 DPI for web quality
    - Font size 9-14 for readability
    - Standard color scheme
    """
    
    AXES = [
        'return_on_equity_pct',
        'return_on_capital_employed_pct',
        'net_profit_margin_pct',
        'debt_to_equity_inverted',  # Inverted so higher is better
        'fcf_score',
        'pat_cagr_5yr_pct',
        'revenue_cagr_5yr_pct',
        'composite_quality_score'
    ]
    
    AXIS_LABELS = [
        'ROE (%)',
        'ROCE (%)',
        'NPM (%)',
        'D/E (Inv)',
        'FCF Score',
        'PAT CAGR (%)',
        'Revenue CAGR (%)',
        'Composite Score'
    ]
    
    def __init__(self, output_dir: str = "reports/radar_charts"):
        """Initialize with output directory."""
        self.output_dir = Path(output_dir)
        self.output_dir.mkdir(parents=True, exist_ok=True)
        logger.info(f"RadarChartGenerator initialized, output: {output_dir}")
    
    def prepare_data_for_radar(self, df: pd.DataFrame) -> pd.DataFrame:
        """
        Prepare data for radar chart visualization.
        
        Creates normalized values (0-100 scale) for each axis.
        """
        df = df.copy()
        
        # Create inverted D/E (lower D/E = higher score)
        df['debt_to_equity_inverted'] = 100 / (1 + df['debt_to_equity'].fillna(0))
        
        # Create FCF score (normalized)
        fcf = df['free_cash_flow_cr'].fillna(0)
        fcf_min, fcf_max = fcf.quantile(0.1), fcf.quantile(0.9)
        df['fcf_score'] = ((fcf - fcf_min) / (fcf_max - fcf_min) * 100).clip(0, 100)
        
        # Ensure composite_quality_score exists
        if 'composite_quality_score' not in df.columns:
            logger.warning("composite_quality_score not found, using 50 as default")
            df['composite_quality_score'] = 50
        
        return df
    
    def generate_radar_chart(self,
                            company_id: int,
                            company_name: str,
                            company_values: pd.Series,
                            peer_avg_values: pd.Series,
                            peer_group: Optional[str] = None) -> str:
        """
        Generate single radar chart for a company.
        
        Args:
            company_id: Company ID
            company_name: Company name
            company_values: Series with values for 8 axes
            peer_avg_values: Series with peer group average values
            peer_group: Name of peer group (or None)
            
        Returns:
            Path to saved PNG file
        """
        # Prepare data
        values = []
        peer_avgs = []
        
        for axis in self.AXES:
            val = company_values.get(axis, 0)
            peer_avg = peer_avg_values.get(axis, 0) if peer_avg_values is not None else 0
            
            # Handle NaN
            val = 0 if pd.isna(val) else float(val)
            peer_avg = 0 if pd.isna(peer_avg) else float(peer_avg)
            
            values.append(val)
            peer_avgs.append(peer_avg)
        
        # Compute angles for radar chart
        angles = np.linspace(0, 2 * np.pi, len(self.AXIS_LABELS), endpoint=False).tolist()
        angles += angles[:1]  # Complete the circle
        
        values += values[:1]
        peer_avgs += peer_avgs[:1]
        
        # Create figure
        fig, ax = plt.subplots(figsize=(8, 8), subplot_kw=dict(projection='polar'))
        
        # Plot company values (filled polygon)
        ax.plot(angles, values, 'o-', linewidth=2, label=company_name, color='#2E86AB')
        ax.fill(angles, values, alpha=0.25, color='#2E86AB')
        
        # Plot peer group average (dashed overlay)
        if peer_group:
            ax.plot(angles, peer_avgs, '--', linewidth=1.5, label=f'{peer_group} Avg', color='#A23B72')
        else:
            ax.plot(angles, peer_avgs, '--', linewidth=1.5, label='Nifty 100 Avg', color='#F18F01')
        
        # Formatting
        ax.set_xticks(angles[:-1])
        ax.set_xticklabels(self.AXIS_LABELS, size=10)
        ax.set_ylim(0, 100)
        ax.set_yticks([20, 40, 60, 80, 100])
        ax.set_yticklabels(['20', '40', '60', '80', '100'], size=8)
        ax.grid(True, linestyle='--', alpha=0.7)
        
        # Legend and title
        title = f"{company_name} (ID: {company_id})"
        if peer_group:
            title += f"\n{peer_group} Peer Comparison"
        
        ax.set_title(title, size=12, pad=20, weight='bold')
        ax.legend(loc='upper right', bbox_to_anchor=(1.3, 1.1), fontsize=9)
        
        # Save
        output_path = self.output_dir / f"{company_id}_radar.png"
        plt.savefig(output_path, dpi=150, bbox_inches='tight')
        plt.close()
        
        logger.info(f"Generated radar chart: {output_path}")
        return str(output_path)
    
    def generate_all_radar_charts(self,
                                 df: pd.DataFrame,
                                 peer_mapping: pd.DataFrame) -> List[str]:
        """
        Generate radar charts for all companies.
        
        Args:
            df: DataFrame with financial data
            peer_mapping: DataFrame with company_id and peer_group_name
            
        Returns:
            List of generated file paths
        """
        df = self.prepare_data_for_radar(df)
        paths_generated = []
        
        for idx, row in df.iterrows():
            company_id = row['company_id']
            company_name = row['company_name']
            
            # Find peer group
            peer_rows = peer_mapping[peer_mapping['company_id'] == company_id]
            
            if len(peer_rows) > 0:
                peer_group = peer_rows.iloc[0]['peer_group_name']
                
                # Get peer group average
                peer_companies = peer_mapping[peer_mapping['peer_group_name'] == peer_group]['company_id'].tolist()
                peer_data = df[df['company_id'].isin(peer_companies)]
                peer_avg = peer_data[self.AXES].mean()
            else:
                peer_group = None
                peer_avg = df[self.AXES].mean()  # Nifty 100 average
            
            # Get company values
            company_values = row[self.AXES]
            
            # Generate chart
            path = self.generate_radar_chart(company_id, company_name, company_values, peer_avg, peer_group)
            paths_generated.append(path)
        
        logger.info(f"Generated {len(paths_generated)} radar charts")
        return paths_generated
